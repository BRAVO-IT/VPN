#!/bin/bash
# SOCKS5 proxy (Dante) multi-IP installer for Ubuntu 20.04+ and Debian 11+
# Based on: https://github.com/BRAVO-IT/VPN (author: BRAVO_IT)
#
# Usage (as root, or through sudo):
#   curl -fsSL YOUR_RAW_URL | sudo bash -s -- IP1,IP2,IP3 [PORT] [USER] [PASSWORD]
#
# Examples:
#   ... | sudo bash -s -- 45.77.60.139,45.77.60.140            # extra IPs, port 8080, user "user", random password
#   ... | sudo bash -s -- 45.77.60.139 8080 user MyPass123     # choose port, user and password
#   ... | sudo bash                                             # no extra IPs: just the main IP
#
# Notes:
#  - List only the EXTRA IPs. The main IP is added automatically.
#  - The IP list you give replaces the previous extra-IP list on this server.
#  - Vultr needs one restart from the Vultr panel after adding a new additional IP.

set -u

if [ "$(id -u)" -ne 0 ]; then
    echo "Please run as root (use sudo)."
    exit 1
fi

EXTRA_RAW="${1:-}"
PORT="${2:-8080}"
PROXY_USER="${3:-user}"
PROXY_PASS="${4:-$(openssl rand -hex 6 2>/dev/null || tr -dc 'a-z0-9' </dev/urandom | head -c 12)}"
[ "$EXTRA_RAW" = "-" ] && EXTRA_RAW=""

# ---- OS check -------------------------------------------------------------
[ -e /etc/os-release ] && . /etc/os-release
MAJOR="${VERSION_ID%%.*}"

case "${ID:-}" in
    ubuntu) [ "${MAJOR:-0}" -ge 20 ] || { echo "Ubuntu 20.04 or newer is required."; exit 1; } ;;
    debian) [ "${MAJOR:-0}" -ge 11 ] || { echo "Debian 11 or newer is required."; exit 1; } ;;
    *)      echo "Unsupported OS: ${PRETTY_NAME:-unknown}. Use Ubuntu 20.04+ or Debian 11+."; exit 1 ;;
esac

# ---- network info ---------------------------------------------------------
IFACE=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')
MAIN_IP=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="src"){print $(i+1); exit}}')
if [ -z "$IFACE" ] || [ -z "$MAIN_IP" ]; then
    echo "Could not detect the network interface or main IP."
    exit 1
fi

# ---- add extra IPs (only when a list was given) ---------------------------
NEEDS_RESTART=0
EXTRA_IPS=""
if [ -n "$EXTRA_RAW" ]; then
    for ip in $(echo "$EXTRA_RAW" | tr ',; ' '\n\n\n'); do
        [ -z "$ip" ] && continue
        if ! echo "$ip" | grep -Eq '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; then
            echo "Invalid IP address: $ip"
            exit 1
        fi
        [ "$ip" = "$MAIN_IP" ] && continue          # main IP is added automatically
        EXTRA_IPS="$EXTRA_IPS $ip"
    done

    if [ -n "$EXTRA_IPS" ]; then
        if command -v netplan >/dev/null 2>&1; then
            {
                echo "network:"
                echo "  version: 2"
                echo "  ethernets:"
                echo "    ${IFACE}:"
                echo "      addresses:"
                for ip in $EXTRA_IPS; do echo "        - ${ip}/32"; done
            } > /etc/netplan/60-extra-ip.yaml
            chmod 600 /etc/netplan/60-extra-ip.yaml
            netplan apply
        else
            echo "netplan not found: adding IPs for this boot only (not persistent)."
            for ip in $EXTRA_IPS; do
                ip addr show dev "$IFACE" | grep -qw "inet ${ip}/32" || ip addr add "${ip}/32" dev "$IFACE"
            done
        fi
        sleep 3

        # can each extra IP send traffic out?
        for ip in $EXTRA_IPS; do
            OUT=$(curl -4 -s -m 10 --interface "$ip" https://ifconfig.me 2>/dev/null)
            if [ "$OUT" != "$ip" ]; then
                echo "Warning: $ip cannot send traffic out yet."
                NEEDS_RESTART=1
            fi
        done
    fi
fi

# every IPv4 now on the interface (main + extras)
ALL_IPS=$(ip -4 -o addr show dev "$IFACE" scope global 2>/dev/null | awk '{print $4}' | cut -d/ -f1)
INTERNAL_LINES=""
EXTERNAL_LINES=""
for ip in $ALL_IPS; do
    INTERNAL_LINES="${INTERNAL_LINES}internal: ${ip} port=${PORT}
"
    EXTERNAL_LINES="${EXTERNAL_LINES}external: ${ip}
"
done

# ---- install --------------------------------------------------------------
export DEBIAN_FRONTEND=noninteractive
apt-get update
apt-get install -y dante-server libpam-pwdfile openssl curl

# stop the package's default (empty-config) instance before we write ours
systemctl stop danted 2>/dev/null || true

# password file (md5-crypt hash)
echo "${PROXY_USER}:$(openssl passwd -1 "$PROXY_PASS")" > /etc/danted.passwd
chmod 600 /etc/danted.passwd

# PAM auth for sockd
cat > /etc/pam.d/sockd <<EOF
auth required pam_pwdfile.so nodelay pwdfile=/etc/danted.passwd
account required pam_permit.so
EOF

# Dante config: listens on every IP, sends out from the same IP the client used
cat > /etc/danted.conf <<EOF
logoutput: syslog

internal.protocol: ipv4
${INTERNAL_LINES}
external.protocol: ipv4
${EXTERNAL_LINES}external.rotation: same-same

socksmethod: pam.any

user.privileged: root
user.notprivileged: nobody

client pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        log: connect disconnect error
}

# deny proxied connections to loopback
socks block {
        from: 0.0.0.0/0 to: 127.0.0.0/8
        log: connect error
}

socks pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        command: bind connect udpassociate
        log: connect disconnect error
}
EOF

# ---- firewall (only if ufw is installed and active) -----------------------
if command -v ufw >/dev/null 2>&1 && ufw status | grep -q "Status: active"; then
    ufw allow "${PORT}/tcp"
fi

# wait for the network (extra IPs) at boot and restart automatically on failure
mkdir -p /etc/systemd/system/danted.service.d
cat > /etc/systemd/system/danted.service.d/override.conf <<'EOF'
[Unit]
After=network-online.target
Wants=network-online.target

[Service]
Restart=on-failure
RestartSec=5
EOF
systemctl daemon-reload

# ---- start service --------------------------------------------------------
systemctl enable danted >/dev/null 2>&1
systemctl restart danted
sleep 2

if ! systemctl is-active --quiet danted; then
    echo "danted failed to start. Check: systemctl status danted  /  journalctl -xeu danted"
    exit 1
fi

# ---- self-test: one request through every IP ------------------------------
echo ""
echo "Your SOCKS5 proxy (same port, user and password on every IP):"
printf "Protocol : SOCKS5\n"
printf "Port     : %s\n" "$PORT"
printf "User     : %s\n" "$PROXY_USER"
printf "Pass     : %s\n" "$PROXY_PASS"
echo ""

FAILED=0
for ip in $ALL_IPS; do
    OUT=$(curl -s -m 20 --socks5-hostname "${ip}:${PORT}" \
            --proxy-user "${PROXY_USER}:${PROXY_PASS}" https://ifconfig.me 2>/dev/null)
    if [ -n "$OUT" ]; then
        echo "Address  : ${ip}   -> exits as ${OUT}   [OK]"
    else
        echo "Address  : ${ip}   -> [FAILED]"
        FAILED=1
    fi
done

echo ""
if [ "$FAILED" -eq 1 ] || [ "$NEEDS_RESTART" -eq 1 ]; then
    echo "Some IPs are not working yet. For a NEW additional IP, restart the server"
    echo "from the Vultr panel (not 'reboot'), wait a minute, then check with:"
    echo "  journalctl -u danted -n 30 --no-pager"
    echo "The settings are saved, so you do not need to run this again after the restart."
fi
echo "Remember to allow TCP port ${PORT} in the Vultr firewall group."
echo "Watch connections live with: journalctl -u danted -f"
