#!/bin/bash
# SOCKS5 proxy (Dante) multi-IP installer with TELEGRAM 1-CLICK COPY & QR CODE
# Based on: https://github.com/BRAVO-IT/VPN (author: BRAVO_IT)

set -u

if [ "$(id -u)" -ne 0 ]; then
    echo "Please run as root (use sudo)."
    exit 1
fi

# ==============================================================================
# TELEGRAM CONFIGURATION
# ==============================================================================
TG_BOT_TOKEN="8328863761:AAF_RDhl9VKzqzvTEzZsh1i3AeAlbwtIImY"      # Example: "7123456789:AAFnk..."
TG_CHAT_ID="1199333247"          # Example: "123456789"
# ==============================================================================

RAW_INPUT="${1:-}"
PORT="${2:-8080}"
DEFAULT_USER_TAG="${3:-USER056}"
DEFAULT_TEAM="${4:-黑虾团队}"
PROXY_USER="root"

# ---- Network Detection ----------------------------------------------------
IFACE=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')
MAIN_IP=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="src"){print $(i+1); exit}}')
if [ -z "$IFACE" ] || [ -z "$MAIN_IP" ]; then
    echo "Could not detect the network interface or main IP."
    exit 1
fi

# Multi-subnet routing support (rp_filter)
sysctl -w net.ipv4.conf.all.rp_filter=2 >/dev/null 2>&1 || true
sysctl -w net.ipv4.conf.default.rp_filter=2 >/dev/null 2>&1 || true
sysctl -w net.ipv4.conf."${IFACE}".rp_filter=2 >/dev/null 2>&1 || true

cat << 'SYSCTL' > /etc/sysctl.d/99-rp-filter.conf
net.ipv4.conf.all.rp_filter = 2
net.ipv4.conf.default.rp_filter = 2
SYSCTL

# Install Dependencies
export DEBIAN_FRONTEND=noninteractive
apt-get update -qq
apt-get install -y -qq dante-server libpam-pwdfile openssl curl iputils-arping qrencode

# ---- Parse IPs, Users, and Teams ------------------------------------------
TAG_PREFIX=$(echo "$DEFAULT_USER_TAG" | sed 's/[0-9]*$//')
TAG_NUM_RAW=$(echo "$DEFAULT_USER_TAG" | grep -o '[0-9]*$')
TAG_NUM_LEN=${#TAG_NUM_RAW}
TAG_NUM=${TAG_NUM_RAW:-0}

declare -A IP_USER_MAP
declare -A IP_TEAM_MAP
EXTRA_IPS=""
COUNTER=0

for item in $(echo "$RAW_INPUT" | tr ',; \t\r' '\n\n\n\n\n'); do
    [ -z "$item" ] && continue

    IFS=':' read -r item_ip item_user item_team <<< "$item"
    if ! echo "$item_ip" | grep -Eq '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; then
        continue
    fi

    if [ -n "$item_user" ]; then
        assigned_user="$item_user"
    else
        curr_num=$((10#$TAG_NUM + COUNTER))
        if [ "$TAG_NUM_LEN" -gt 0 ]; then
            assigned_user=$(printf "%s%0${TAG_NUM_LEN}d" "$TAG_PREFIX" "$curr_num")
        else
            assigned_user="${TAG_PREFIX}${curr_num}"
        fi
    fi

    assigned_team="${item_team:-$DEFAULT_TEAM}"
    IP_USER_MAP["$item_ip"]="$assigned_user"
    IP_TEAM_MAP["$item_ip"]="$assigned_team"

    [ "$item_ip" = "$MAIN_IP" ] && continue

    EXTRA_IPS="$EXTRA_IPS $item_ip"
    ip addr show dev "$IFACE" | grep -qw "inet ${item_ip}/32" || ip addr add "${item_ip}/32" dev "$IFACE"
    arping -U -c 2 -w 1 -I "$IFACE" "$item_ip" >/dev/null 2>&1 || true

    COUNTER=$((COUNTER + 1))
done

# Persist in Netplan
if [ -n "$EXTRA_IPS" ] && command -v netplan >/dev/null 2>&1; then
    {
        echo "network:"
        echo "  version: 2"
        echo "  ethernets:"
        echo "    ${IFACE}:"
        echo "      addresses:"
        for ip in $EXTRA_IPS; do echo "        - ${ip}/32"; done
    } > /etc/netplan/60-extra-ip.yaml
    chmod 600 /etc/netplan/60-extra-ip.yaml
    netplan apply >/dev/null 2>&1 || true
fi

# Multi-instance systemd service
systemctl stop danted >/dev/null 2>&1 || true
systemctl disable danted >/dev/null 2>&1 || true

cat << 'EOF' > /etc/systemd/system/danted@.service
[Unit]
Description=SOCKS5 proxy daemon for %I
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
ExecStart=/usr/sbin/danted -f /etc/danted-%i.conf -p /run/danted-%i.pid
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF
systemctl daemon-reload

# Save TG Credentials and nodes info
cat << EOF > /etc/proxy-telegram.conf
TG_BOT_TOKEN="${TG_BOT_TOKEN}"
TG_CHAT_ID="${TG_CHAT_ID}"
PORT="${PORT}"
EOF

mkdir -p /etc/proxy-nodes
ALL_IPS=$(ip -4 -o addr show dev "$IFACE" scope global 2>/dev/null | awk '{print $4}' | cut -d/ -f1)
NODE_LIST=()

for ip in $ALL_IPS; do
    NODE_PASS=$(openssl rand -hex 6 2>/dev/null || tr -dc 'a-z0-9' </dev/urandom | head -c 12)

    echo "${PROXY_USER}:$(openssl passwd -1 "$NODE_PASS")" > "/etc/danted-${ip}.passwd"
    chmod 600 "/etc/danted-${ip}.passwd"

    cat << EOF > "/etc/pam.d/sockd-${ip}"
auth required pam_pwdfile.so nodelay pwdfile=/etc/danted-${ip}.passwd
account required pam_permit.so
EOF

    cat << EOF > "/etc/danted-${ip}.conf"
logoutput: syslog

internal.protocol: ipv4
internal: ${ip} port=${PORT}

external.protocol: ipv4
external: ${ip}

socksmethod: pam.any

user.privileged: root
user.notprivileged: nobody

client pass { from: 0.0.0.0/0 to: 0.0.0.0/0 log: error }
socks block { from: 0.0.0.0/0 to: 127.0.0.0/8 }
socks pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        pamservicename: sockd-${ip}
        command: bind connect udpassociate
        log: error
}
EOF

    systemctl enable "danted@${ip}" >/dev/null 2>&1
    systemctl restart "danted@${ip}"

    curr_user="${IP_USER_MAP[$ip]:-$DEFAULT_USER_TAG}"
    curr_team="${IP_TEAM_MAP[$ip]:-$DEFAULT_TEAM}"

    # Save to node info file for change-pass
    echo "$curr_user" > "/etc/proxy-nodes/${ip}.info"
    echo "$curr_team" >> "/etc/proxy-nodes/${ip}.info"

    NODE_LIST+=("${ip}:${PORT}:${PROXY_USER}:${NODE_PASS}:${curr_user}:${curr_team}")
done

# Firewall
if command -v ufw >/dev/null 2>&1 && ufw status | grep -q "Status: active"; then
    ufw allow "${PORT}/tcp" >/dev/null 2>&1 || true
fi

sleep 2

# Output & Send to Telegram
echo ""
echo "================================================================"
echo "          YOUR SOCKS5 PROXIES (1-CLICK COPYABLE)                "
echo "================================================================"

for node in "${NODE_LIST[@]}"; do
    node_ip=$(echo "$node" | cut -d: -f1)
    node_port=$(echo "$node" | cut -d: -f2)
    node_user=$(echo "$node" | cut -d: -f3)
    node_pass=$(echo "$node" | cut -d: -f4)
    node_tag=$(echo "$node" | cut -d: -f5)
    node_team=$(echo "$node" | cut -d: -f6)

    AUTH_B64=$(echo -n "${node_user}:${node_pass}" | base64 | tr -d '\n')
    SOCKS_URL="socks://${AUTH_B64}@${node_ip}:${node_port}#${node_tag}"

    CAPTION="\`${SOCKS_URL}\`

${node_team}

${node_tag}
-Protocol : \`SOCKS5\`
-IP public : \`${node_ip}\`
-Port         : \`${node_port}\`
-User         : \`${node_user}\`
-Pass         : \`${node_pass}\`"

    echo "$CAPTION"
    echo "----------------------------------------------------------------"

    qrencode -s 8 -m 2 -o "/tmp/qr_${node_ip}.png" "$SOCKS_URL"

    if [ -n "$TG_BOT_TOKEN" ] && [ "$TG_BOT_TOKEN" != "YOUR_BOT_TOKEN_HERE" ]; then
        curl -s -X POST "https://api.telegram.org/bot${TG_BOT_TOKEN}/sendPhoto" \
            -F chat_id="${TG_CHAT_ID}" \
            -F photo="@/tmp/qr_${node_ip}.png" \
            -F parse_mode="Markdown" \
            -F caption="$CAPTION" >/dev/null 2>&1
    fi

    rm -f "/tmp/qr_${node_ip}.png"
done

echo "================================================================"
if [ -n "$TG_BOT_TOKEN" ] && [ "$TG_BOT_TOKEN" != "YOUR_BOT_TOKEN_HERE" ]; then
    echo "✅ Sent 1-click copyable QR codes and details to Telegram!"
fi
echo "================================================================"
