cat << 'EOF' > /usr/local/bin/setup-socks
#!/bin/bash
set -u

if [ "$(id -u)" -ne 0 ]; then
    echo "Please run as root (use sudo)."
    exit 1
fi

INPUT_SOURCE="${1:-}"
PORT="${2:-8080}"
PROXY_USER="${3:-user}"
PROXY_PASS="${4:-fb93e1f6bb5e}"

if [ -z "$INPUT_SOURCE" ]; then
    echo "Usage:"
    echo "  setup-socks /path/to/ips.txt [PORT] [USER] [PASS]"
    echo "  setup-socks \"IP1,IP2,IP3...\" [PORT] [USER] [PASS]"
    exit 1
fi

# If argument is a file, read from file; otherwise treat as raw string
if [ -f "$INPUT_SOURCE" ]; then
    RAW_DATA=$(cat "$INPUT_SOURCE")
else
    RAW_DATA="$INPUT_SOURCE"
fi

# Detect network interface and main IP
IFACE=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')
MAIN_IP=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="src"){print $(i+1); exit}}')

if [ -z "$IFACE" ] || [ -z "$MAIN_IP" ]; then
    echo "Error: Could not detect network interface or main IP."
    exit 1
fi

echo "=================================================="
echo "Detected Main IP: $MAIN_IP on $IFACE"
echo "Port: $PORT | User: $PROXY_USER | Pass: $PROXY_PASS"
echo "=================================================="

# 1. Fix Linux rp_filter for cross-subnet IPs permanently
sysctl -w net.ipv4.conf.all.rp_filter=2 >/dev/null
sysctl -w net.ipv4.conf.default.rp_filter=2 >/dev/null
sysctl -w net.ipv4.conf."${IFACE}".rp_filter=2 >/dev/null

cat << 'SYSCTL' > /etc/sysctl.d/99-rp-filter.conf
net.ipv4.conf.all.rp_filter = 2
net.ipv4.conf.default.rp_filter = 2
SYSCTL

# 2. Install required packages
export DEBIAN_FRONTEND=noninteractive
apt-get update -qq
apt-get install -y -qq dante-server libpam-pwdfile openssl curl iputils-arping

# 3. Add all secondary IPs and send gratuitous ARP
EXTRA_IPS=""
COUNT=0

for ip in $(echo "$RAW_DATA" | tr ',; \t\r' '\n\n\n\n\n'); do
    [ -z "$ip" ] && continue
    if ! echo "$ip" | grep -Eq '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; then
        continue
    fi
    [ "$ip" = "$MAIN_IP" ] && continue

    EXTRA_IPS="$EXTRA_IPS $ip"
    COUNT=$((COUNT + 1))

    # Add to interface immediately
    ip addr show dev "$IFACE" | grep -qw "inet ${ip}/32" || ip addr add "${ip}/32" dev "$IFACE"

    # Gratuitous ARP to tell Vultr's switch immediately
    arping -U -c 2 -w 1 -I "$IFACE" "$ip" >/dev/null 2>&1 || true
done

echo "Configured $COUNT secondary IP(s)."

# Persist to Netplan across reboots
if [ -n "$EXTRA_IPS" ] && command -v netplan >/dev/null 2>&1; then
    {
        echo "network:"
        echo "  version: 2"
        echo "  ethernets:"
        echo "    ${IFACE}:"
        echo "      addresses:"
        for ip in $EXTRA_IPS; do echo "        - ${ip}/32"; done
    } > /etc/netplan/60-extra-ip.yaml
    chmod 600 /etc/netplan/60-extra-ip.yaml
    netplan apply
fi

# 4. Generate Dante Config for all IPs (main + extras)
ALL_IPS=$(ip -4 -o addr show dev "$IFACE" scope global 2>/dev/null | awk '{print $4}' | cut -d/ -f1)

INTERNAL_LINES=""
EXTERNAL_LINES=""
for ip in $ALL_IPS; do
    INTERNAL_LINES="${INTERNAL_LINES}internal: ${ip} port=${PORT}
"
    EXTERNAL_LINES="${EXTERNAL_LINES}external: ${ip}
"
done

echo "${PROXY_USER}:$(openssl passwd -1 "$PROXY_PASS")" > /etc/danted.passwd
chmod 600 /etc/danted.passwd

cat > /etc/pam.d/sockd << 'PAM'
auth required pam_pwdfile.so nodelay pwdfile=/etc/danted.passwd
account required pam_permit.so
PAM

cat > /etc/danted.conf << DANTE
logoutput: syslog

internal.protocol: ipv4
${INTERNAL_LINES}
external.protocol: ipv4
${EXTERNAL_LINES}external.rotation: same-same

socksmethod: pam.any

user.privileged: root
user.notprivileged: nobody

client pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        log: error
}

socks block {
        from: 0.0.0.0/0 to: 127.0.0.0/8
        log: error
}

socks pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        command: bind connect udpassociate
        log: error
}
DANTE

# Open UFW firewall if active
if command -v ufw >/dev/null 2>&1 && ufw status | grep -q "Status: active"; then
    ufw allow "${PORT}/tcp" >/dev/null 2>&1
fi

# Systemd override to auto restart
mkdir -p /etc/systemd/system/danted.service.d
cat > /etc/systemd/system/danted.service.d/override.conf << 'SERVICE'
[Unit]
After=network-online.target
Wants=network-online.target

[Service]
Restart=on-failure
RestartSec=5
SERVICE
systemctl daemon-reload

systemctl restart danted
sleep 2

# 5. Test all IPs
echo "=================================================="
echo "Testing connections through each IP..."
echo "=================================================="

TOTAL_OK=0
TOTAL_FAIL=0

for ip in $ALL_IPS; do
    OUT=$(curl -s -m 8 --socks5-hostname "${ip}:${PORT}" \
            --proxy-user "${PROXY_USER}:${PROXY_PASS}" https://ifconfig.me 2>/dev/null || true)
    if [ "$OUT" = "$ip" ]; then
        echo "IP: ${ip} -> exits as ${OUT} [OK]"
        TOTAL_OK=$((TOTAL_OK + 1))
    else
        echo "IP: ${ip} -> [FAILED]"
        TOTAL_FAIL=$((TOTAL_FAIL + 1))
    fi
done

echo "=================================================="
echo "Finished! [OK]: $TOTAL_OK | [FAILED]: $TOTAL_FAIL"
echo "=================================================="
EOF

chmod +x /usr/local/bin/setup-socks
