#!/bin/bash
# SOCKS5 proxy (Dante) installer v3 for Ubuntu 20.04+ and Debian 11+
# Based on: https://github.com/BRAVO-IT/VPN (author: BRAVO_IT)
#
# Usage (run as root or with sudo):
#   sudo bash socks5-install-v3.sh                       # port 8080, user "user", random password
#   sudo bash socks5-install-v3.sh 8080 user mypassword  # custom port / user / password
#
# Safe to run again: it rewrites the config and restarts the service.

set -u

if [ "$(id -u)" -ne 0 ]; then
    echo "Please run as root: sudo bash $0"
    exit 1
fi

PORT="${1:-8080}"
PROXY_USER="${2:-user}"
PROXY_PASS="${3:-$(openssl rand -hex 6 2>/dev/null || tr -dc 'a-z0-9' </dev/urandom | head -c 12)}"

# ---- OS check -------------------------------------------------------------
[ -e /etc/os-release ] && . /etc/os-release
MAJOR="${VERSION_ID%%.*}"

case "${ID:-}" in
    ubuntu) [ "${MAJOR:-0}" -ge 20 ] || { echo "Ubuntu 20.04 or newer is required."; exit 1; } ;;
    debian) [ "${MAJOR:-0}" -ge 11 ] || { echo "Debian 11 or newer is required."; exit 1; } ;;
    *)      echo "Unsupported OS: ${PRETTY_NAME:-unknown}. Use Ubuntu 20.04+ or Debian 11+."; exit 1 ;;
esac

# ---- network info ---------------------------------------------------------
IFACE=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')
LOCAL_IP=$(ip -4 addr show "$IFACE" 2>/dev/null | awk '/inet /{print $2}' | cut -d/ -f1 | head -1)
if [ -z "$IFACE" ] || [ -z "$LOCAL_IP" ]; then
    echo "Could not detect the network interface or its IPv4 address."
    exit 1
fi

get_external_address() {
    local addr
    addr=$( timeout 4 curl -4 -s https://ifconfig.me \
         || timeout 4 curl -4 -s http://ident.me \
         || timeout 4 curl -4 -s http://ipecho.net/plain )
    [ -z "$addr" ] && addr="<this server IP address>"
    echo "$addr"
}

# ---- install --------------------------------------------------------------
export DEBIAN_FRONTEND=noninteractive
apt-get update
apt-get install -y dante-server libpam-pwdfile openssl curl

# stop the package's default (empty-config) instance before we write ours
systemctl stop danted 2>/dev/null || true

# password file (md5-crypt hash)
echo "${PROXY_USER}:$(openssl passwd -1 "$PROXY_PASS")" > /etc/danted.passwd
chmod 600 /etc/danted.passwd

# PAM auth for sockd
cat > /etc/pam.d/sockd <<EOF
auth required pam_pwdfile.so nodelay pwdfile=/etc/danted.passwd
account required pam_permit.so
EOF

# Dante config: IPv4 only (IPv6 outbound caused failed requests from phone apps)
cat > /etc/danted.conf <<EOF
logoutput: syslog

internal.protocol: ipv4
internal: ${IFACE} port=${PORT}
external.protocol: ipv4
external: ${IFACE}

socksmethod: pam.any

user.privileged: root
user.notprivileged: nobody

client pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        log: connect disconnect error
}

# deny proxied connections to loopback
socks block {
        from: 0.0.0.0/0 to: 127.0.0.0/8
        log: connect error
}

socks pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        command: bind connect udpassociate
        log: connect disconnect error
}
EOF

# ---- firewall (only if ufw is installed and active) -----------------------
if command -v ufw >/dev/null 2>&1 && ufw status | grep -q "Status: active"; then
    ufw allow "${PORT}/tcp"
fi

# ---- start service --------------------------------------------------------
systemctl enable danted >/dev/null 2>&1
systemctl restart danted
sleep 2

if ! systemctl is-active --quiet danted; then
    echo "danted failed to start. Check: systemctl status danted  /  journalctl -xeu danted"
    exit 1
fi

# ---- self-test (same kind of request a phone app makes) -------------------
CODE=$(curl -s -m 20 -o /dev/null -w '%{http_code}' \
        --socks5-hostname "${LOCAL_IP}:${PORT}" \
        --proxy-user "${PROXY_USER}:${PROXY_PASS}" \
        https://www.google.com 2>/dev/null)

echo ""
echo "Your SOCKS5 proxy:"
printf "Protocol : SOCKS5\n"
printf "Address  : %s\n" "$(get_external_address)"
printf "Port     : %s\n" "$PORT"
printf "User     : %s\n" "$PROXY_USER"
printf "Pass     : %s\n" "$PROXY_PASS"
echo ""

case "$CODE" in
    200|301|302) echo "Self-test: OK (HTTP $CODE through the proxy)" ;;
    *)           echo "Self-test: FAILED (got '${CODE:-no response}'). Check: journalctl -u danted -n 30 --no-pager" ;;
esac

echo ""
echo "Remember to allow TCP port ${PORT} in the Vultr firewall group."
echo "Watch connections live with: journalctl -u danted -f"
