#!/bin/bash
# SOCKS5 proxy (Dante) multi-IP installer with UNIQUE PASSWORDS per IP
# Based on: https://github.com/BRAVO-IT/VPN (author: BRAVO_IT)

set -u

if [ "$(id -u)" -ne 0 ]; then
    echo "Please run as root (use sudo)."
    exit 1
fi

EXTRA_RAW="${1:-}"
PORT="${2:-8080}"
[ "$EXTRA_RAW" = "-" ] && EXTRA_RAW=""

# ---- network info ---------------------------------------------------------
IFACE=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')
MAIN_IP=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="src"){print $(i+1); exit}}')
if [ -z "$IFACE" ] || [ -z "$MAIN_IP" ]; then
    echo "Could not detect the network interface or main IP."
    exit 1
fi

# Kernel multi-subnet support (rp_filter)
sysctl -w net.ipv4.conf.all.rp_filter=2 >/dev/null 2>&1 || true
sysctl -w net.ipv4.conf.default.rp_filter=2 >/dev/null 2>&1 || true
sysctl -w net.ipv4.conf."${IFACE}".rp_filter=2 >/dev/null 2>&1 || true

cat << 'SYSCTL' > /etc/sysctl.d/99-rp-filter.conf
net.ipv4.conf.all.rp_filter = 2
net.ipv4.conf.default.rp_filter = 2
SYSCTL

# Install dependencies
export DEBIAN_FRONTEND=noninteractive
apt-get update -qq
apt-get install -y -qq dante-server libpam-pwdfile openssl curl iputils-arping

# Add extra IPs
EXTRA_IPS=""
if [ -n "$EXTRA_RAW" ]; then
    for ip in $(echo "$EXTRA_RAW" | tr ',; \t\r' '\n\n\n\n\n'); do
        [ -z "$ip" ] && continue
        if ! echo "$ip" | grep -Eq '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; then
            continue
        fi
        [ "$ip" = "$MAIN_IP" ] && continue

        EXTRA_IPS="$EXTRA_IPS $ip"
        ip addr show dev "$IFACE" | grep -qw "inet ${ip}/32" || ip addr add "${ip}/32" dev "$IFACE"
        arping -U -c 2 -w 1 -I "$IFACE" "$ip" >/dev/null 2>&1 || true
    done

    # Persist in Netplan
    if [ -n "$EXTRA_IPS" ] && command -v netplan >/dev/null 2>&1; then
        {
            echo "network:"
            echo "  version: 2"
            echo "  ethernets:"
            echo "    ${IFACE}:"
            echo "      addresses:"
            for ip in $EXTRA_IPS; do echo "        - ${ip}/32"; done
        } > /etc/netplan/60-extra-ip.yaml
        chmod 600 /etc/netplan/60-extra-ip.yaml
        netplan apply >/dev/null 2>&1 || true
    fi
fi

# Gather all IPs
ALL_IPS=$(ip -4 -o addr show dev "$IFACE" scope global 2>/dev/null | awk '{print $4}' | cut -d/ -f1)

INTERNAL_LINES=""
EXTERNAL_LINES=""
for ip in $ALL_IPS; do
    INTERNAL_LINES="${INTERNAL_LINES}internal: ${ip} port=${PORT}
"
    EXTERNAL_LINES="${EXTERNAL_LINES}external: ${ip}
"
done

# Stop danted before updating
systemctl stop danted 2>/dev/null || true

# Generate UNIQUE user and UNIQUE random password for EACH IP
> /etc/danted.passwd
NODE_LIST=()
INDEX=1

for ip in $ALL_IPS; do
    NODE_USER="user${INDEX}"
    NODE_PASS=$(openssl rand -hex 6 2>/dev/null || tr -dc 'a-z0-9' </dev/urandom | head -c 12)
    
    # Save user & unique password to PAM password file
    echo "${NODE_USER}:$(openssl passwd -1 "$NODE_PASS")" >> /etc/danted.passwd
    
    # Save to node list
    NODE_LIST+=("${ip}:${PORT}:${NODE_USER}:${NODE_PASS}")
    INDEX=$((INDEX + 1))
done
chmod 600 /etc/danted.passwd

# PAM configuration
cat > /etc/pam.d/sockd << 'EOF'
auth required pam_pwdfile.so nodelay pwdfile=/etc/danted.passwd
account required pam_permit.so
EOF

# Dante configuration
cat > /etc/danted.conf << EOF
logoutput: syslog

internal.protocol: ipv4
${INTERNAL_LINES}
external.protocol: ipv4
${EXTERNAL_LINES}external.rotation: same-same

socksmethod: pam.any

user.privileged: root
user.notprivileged: nobody

client pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        log: error
}

socks block {
        from: 0.0.0.0/0 to: 127.0.0.0/8
}

socks pass {
        from: 0.0.0.0/0 to: 0.0.0.0/0
        command: bind connect udpassociate
        log: error
}
EOF

# Firewall
if command -v ufw >/dev/null 2>&1 && ufw status | grep -q "Status: active"; then
    ufw allow "${PORT}/tcp" >/dev/null 2>&1 || true
fi

# Systemd override for auto-recovery
mkdir -p /etc/systemd/system/danted.service.d
cat > /etc/systemd/system/danted.service.d/override.conf << 'EOF'
[Unit]
After=network-online.target
Wants=network-online.target

[Service]
Restart=on-failure
RestartSec=5
EOF
systemctl daemon-reload

systemctl enable danted >/dev/null 2>&1
systemctl restart danted
sleep 2

# Output and Self-test
echo ""
echo "================================================================"
echo "          YOUR SOCKS5 PROXIES (DIFFERENT PASSWORDS)             "
echo "================================================================"
echo "Proxy List (IP:Port:User:Pass):"
for node in "${NODE_LIST[@]}"; do
    echo "  $node"
done
echo "----------------------------------------------------------------"
echo "Testing Live Nodes:"

for node in "${NODE_LIST[@]}"; do
    node_ip=$(echo "$node" | cut -d: -f1)
    node_port=$(echo "$node" | cut -d: -f2)
    node_user=$(echo "$node" | cut -d: -f3)
    node_pass=$(echo "$node" | cut -d: -f4)

    OUT=$(curl -s -m 8 --socks5-hostname "${node_ip}:${node_port}" \
            --proxy-user "${node_user}:${node_pass}" https://ifconfig.me 2>/dev/null || true)
    if [ "$OUT" = "$node_ip" ]; then
        echo "Address  : ${node_ip} (${node_user})   -> exits as ${OUT}   [OK]"
    else
        echo "Address  : ${node_ip} (${node_user})   -> [FAILED]"
    fi
done
echo "================================================================"
