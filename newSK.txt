cat << 'EOF' > /usr/local/bin/setup-multi-socks5.sh
#!/bin/bash
set -u

if [ "$(id -u)" -ne 0 ]; then
    echo "Error: Please run as root (use sudo)."
    exit 1
fi

IP_INPUT="${1:-}"
PORT="${2:-8080}"
PROXY_USER="${3:-user}"
PROXY_PASS="${4:-fb93e1f6bb5e}"

if [ -z "$IP_INPUT" ]; then
    echo "Usage: $0 \"IP1,IP2,IP3,...\" [PORT] [USER] [PASSWORD]"
    exit 1
fi

# Detect main network interface and main IP
IFACE=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')
MAIN_IP=$(ip -4 route get 8.8.8.8 | awk '{for(i=1;i<=NF;i++) if($i=="src"){print $(i+1); exit}}')

if [ -z "$IFACE" ] || [ -z "$MAIN_IP" ]; then
    echo "Error: Could not detect network interface or main IP."
    exit 1
fi

echo "=================================================="
echo "Main IP       : $MAIN_IP on $IFACE"
echo "Port          : $PORT"
echo "Username      : $PROXY_USER"
echo "Password      : $PROXY_PASS"
echo "=================================================="

# 1. Permanently fix Linux rp_filter for multiple subnets
echo "[1/5] Applying kernel network settings (rp_filter=2)..."
sysctl -w net.ipv4.conf.all.rp_filter=2 >/dev/null
sysctl -w net.ipv4.conf.default.rp_filter=2 >/dev/null
sysctl -w net.ipv4.conf."${IFACE}".rp_filter=2 >/dev/null

cat << 'SYSCTL' > /etc/sysctl.d/99-rp-filter.conf
net.ipv4.conf.all.rp_filter = 2
net.ipv4.conf.default.rp_filter = 2
SYSCTL

# 2. Install dependencies
echo "[2/5] Installing Dante & arping..."
export DEBIAN_FRONTEND=noninteractive
apt-get update -qq
apt-get install -y -qq dante-server libpam-pwdfile openssl curl iputils-arping

# 3. Process all IPs and notify Vultr with gratuitous ARP
echo "[3/5] Binding IPs and announcing to Vultr gateway..."
EXTRA_IPS=""
TOTAL_COUNT=0

# Parse comma, space, or newline separated list
for ip in $(echo "$IP_INPUT" | tr ',; \t' '\n\n\n\n'); do
    [ -z "$ip" ] && continue
    # Basic IP validation
    if ! echo "$ip" | grep -Eq '^([0-9]{1,3}\.){3}[0-9]{1,3}$'; then
        echo "Skipping invalid IP format: $ip"
        continue
    fi
    [ "$ip" = "$MAIN_IP" ] && continue

    EXTRA_IPS="$EXTRA_IPS $ip"
    TOTAL_COUNT=$((TOTAL_COUNT + 1))

    # Add IP to interface immediately
    ip addr show dev "$IFACE" | grep -qw "inet ${ip}/32" || ip addr add "${ip}/32" dev "$IFACE"

    # Send Gratuitous ARP so Vultr router instantly routes traffic without reboot
    arping -U -c 2 -w 1 -I "$IFACE" "$ip" >/dev/null 2>&1 || true
done

echo "Added $TOTAL_COUNT secondary IP(s)."

# Save permanently to Netplan
if [ -n "$EXTRA_IPS" ] && command -v netplan >/dev/null 2>&1; then
    {
        echo "network:"
        echo "  version: 2"
        echo "  ethernets:"
        echo "    ${IFACE}:"
        echo "      addresses:"
        for ip in $EXTRA_IPS; do echo "        - ${ip}/32"; done
    } > /etc/netplan/60-extra-ip.yaml
    chmod 600 /etc/netplan/60-extra-ip.yaml
    netplan apply
fi

# 4. Configure Dante SOCKS5
echo "[4/5] Generating Dante SOCKS5 configuration..."
ALL_IPS=$(ip -4 -o addr show dev "$IFACE" scope global | awk '{print $4}' | cut -d/ -f1)

INTERNAL_LINES=""
EXTERNAL_LINES=""
for ip in $ALL_IPS; do
    INTERNAL_LINES="${INTERNAL_LINES}internal: ${ip} port=${PORT}\n"
    EXTERNAL_LINES="${EXTERNAL_LINES}external: ${ip}\n"
done

# Password configuration
echo "${PROXY_USER}:$(openssl passwd -1 "$PROXY_PASS")" > /etc/danted.passwd
chmod 600 /etc/danted.passwd

cat > /etc/pam.d/sockd << 'PAM'
auth required pam_pwdfile.so nodelay pwdfile=/etc/danted.passwd
account required pam_permit.so
PAM

cat > /etc/danted.conf << DANTE
logoutput: syslog

internal.protocol: ipv4
$(echo -e "$INTERNAL_LINES")
external.protocol: ipv4
$(echo -e "$EXTERNAL_LINES")
external.rotation: same-same

socksmethod: pam.any

user.privileged: root
user.notprivileged: nobody

client pass {
    from: 0.0.0.0/0 to: 0.0.0.0/0
    log: error
}

socks block {
    from: 0.0.0.0/0 to: 127.0.0.0/8
}

socks pass {
    from: 0.0.0.0/0 to: 0.0.0.0/0
    command: bind connect udpassociate
    log: error
}
DANTE

# Firewall (UFW) if enabled
if command -v ufw >/dev/null 2>&1 && ufw status | grep -q "Status: active"; then
    ufw allow "${PORT}/tcp" >/dev/null 2>&1
fi

systemctl restart danted
sleep 2

# 5. Test all IPs
echo "[5/5] Testing all IPs..."
echo "=================================================="
PASSED=0
FAILED=0

for ip in $ALL_IPS; do
    OUT=$(curl -s -m 6 --socks5-hostname "${ip}:${PORT}" --proxy-user "${PROXY_USER}:${PROXY_PASS}" https://ifconfig.me 2>/dev/null || true)
    if [ "$OUT" = "$ip" ]; then
        echo "Address: ${ip} -> exits as ${OUT} [OK]"
        PASSED=$((PASSED + 1))
    else
        echo "Address: ${ip} -> [FAILED]"
        FAILED=$((FAILED + 1))
    fi
done

echo "=================================================="
echo "Done! Total Working: $PASSED | Total Failed: $FAILED"
echo "Dante SOCKS5 is active and saved permanently."
EOF

chmod +x /usr/local/bin/setup-multi-socks5.sh
